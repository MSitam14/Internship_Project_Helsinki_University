# GRID PARAMETERS --------------------------------------- #

explicit_grid = False
    # Default : False
    # Possible values :
        # True : Uses an explicit grid. A 3D numpy array is created. Useful for visualisation and analysis of the grid.
        # False : Uses an implicit grid. A 1D numpy array is created. Only the grid coordinates are stored.
    # Note : An explicit grid is a grid that is defined by the user, an implicit grid is a grid that is defined by the program

atom_type = SYBYL
    # Default : normal
    # Possible values :
        # normal : Uses the normal atom types
        # SYBYL : Uses the SYBYL atom types
    # Note : The atom types used for the comparison

grid_spacing = 0.35
	# Default : 0.25
	# Possible values :
		# Any positive value, represents the angstrom distance between two points of the grid
	# Note : Small values makes the comparison slower
	# Note : It may be interesting to use a value similar to the structure resolution
	# Warning : Distances inferiors to 0.2 A are time and memory consuming

grid_padding = 1.0
    # Default : 3.0
    # Possible values :
        # Any positive value, represents the angstrom distance between the grid edge and the structure
    # Note : Big values use more memory and time

align_principal_axes = True
    # Default : False
    # Possible values :
        # True : Aligns the principal axes of the structures to the origin of the coordinate system
        # False : Does not align the principal axes of the structures
    # Note : The principal axes are used to align the structures therefore to optimize the size of the grid

# COMPARISON PARAMETERS --------------------------------- #

consider_elements = True
	# Default : True
	# Possible values :
		# True : Compares by distinguishing between different types of atoms
		# False : Compares only the volumes of the structures, without considering the types of elements
	# Note : Comparing the volumes only increases the similarity score, often generating a stronger tree

comparison_normalisation = Min
	# Default : Max
	# Possible values :
		# Max : Normalizes the similarity score with the maximal number of occupied points between the grids
		# Min : Normalizes the similarity score with the minimal number of occupied points between the grids
	# Note :

grid_geometry = Sphere
	# Default : Sphere
	# Possible values :
		# Sphere : Uses the default sphere formula for the euclidean geometry
		# Taxicab : Uses the taxicab geometry, where the distance between 2 points is the sum of the absolute differences of their coordinates
		# Uniform : Applies an uniform scaling in the euclidean geometry
	# Note : Approximation used for the Van der Waals radius determination

delete_grid = False
    # Default : True
    # Possible values :
        # True : Deletes the grid after the comparison
        # False : Keeps the grid after the comparison
    # Note : The grid is used to compare the structures
