# METHODS PARAMETERS ------------------------------------ #

run_comparison = False
	# Default : True
	# Possible values :
		# True : Runs a structure comparison analysis
		# False : Do not run a structure comparison analysis

run_hotspot = True
	# Default : True
	# Possible values :
		# True : Runs a structure hotspot analysis
		# False : Do not run a structure hotspot analysis

# ------------------------------------------------------- #

# GRID PARAMETERS --------------------------------------- #

explicit_grid = False
    # Default : False
    # Possible values :
        # True : Uses an explicit grid. A 3D numpy array is created. Useful for visualisation and analysis of the grid.
        # False : Uses an implicit grid. A 1D numpy array is created. Only the grid coordinates are stored.
    # Note : An explicit grid is a grid that is defined by the user, an implicit grid is a grid that is defined by the program

atom_type = SYBYL
    # Default : normal
    # Possible values :
        # normal : Uses the normal atom types
        # SYBYL : Uses the SYBYL atom types
    # Note : The atom types used for the comparison

grid_spacing = 0.35
	# Default : 0.25
	# Possible values :
		# Any positive value, represents the angstrom distance between two points of the grid
	# Note : Small values makes the comparison slower
	# Note : It may be interesting to use a value similar to the structure resolution
	# Warning : Distances inferiors to 0.2 A are time and memory consuming

grid_padding = 1.0
    # Default : 3.0
    # Possible values :
        # Any positive value, represents the angstrom distance between the grid edge and the structure
    # Note : Big values use more memory and time

align_principal_axes = True
    # Default : False
    # Possible values :
        # True : Aligns the principal axes of the structures to the origin of the coordinate system
        # False : Does not align the principal axes of the structures
    # Note : The principal axes are used to align the structures therefore to optimize the size of the grid

# ------------------------------------------------------- #

# POCKET PARAMETERS ------------------------------------- #

pocket_res_name = False
    # Default : False
	# Possible values :
	    # False : if dataset_type = proteins or dockings or if your pockets already prepared
	    # List of 3 letter residue names : pocket will be created around them
	# Note : 3 letter residue name, if you already have pocket pdb files ready, put False

pocket_res_id = False
    # Default : False
    # Possible values :
        # False : if dataset_type = proteins or dockings, or if you want to create pockets around every res_name
        # any residue number : pocket will be created around it
    # Note : residue number, if you already have pocket pdb files ready, put False

lig_chain = False
    # Default : False
    # Possible values :
        # any chain id : chain containing the ligand to create the pocket

pocket_size = 5
    # Default : 4.5
    # Possible values :
        # Any positive value, represents the size of the pocket in angstrom
    # Note : 4.5 angstroms is the default value for a pocket-size

# ------------------------------------------------------- #
