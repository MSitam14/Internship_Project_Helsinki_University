# COMPARISON PARAMETERS --------------------------------- #

consider_elements = True
	# Default : True
	# Possible values :
		# True : Compares by distinguishing between different types of atoms
		# False : Compares only the volumes of the structures, without considering the types of elements
	# Note : Comparing the volumes only increases the similarity score, often generating a stronger tree

comparison_normalisation = Min
	# Default : Max
	# Possible values :
		# Max : Normalizes the similarity score with the maximal number of occupied points between the grids
		# Min : Normalizes the similarity score with the minimal number of occupied points between the grids
	# Note :

grid_geometry = Sphere
	# Default : Sphere
	# Possible values :
		# Sphere : Uses the default sphere formula for the euclidean geometry
		# Taxicab : Uses the taxicab geometry, where the distance between 2 points is the sum of the absolute differences of their coordinates
		# Uniform : Applies an uniform scaling in the euclidean geometry
	# Note : Approximation used for the Van der Waals radius determination

delete_grid = False
    # Default : True
    # Possible values :
        # True : Deletes the grid after the comparison
        # False : Keeps the grid after the comparison
    # Note : The grid is used to compare the structures

# PYMOL SESSION --------------------------------------- #

dataset_status = 2
	# Default : 0
	# Possible values :
	    # 0 : The dataset is not cleaned and proteins are not structurally aligned
        # 1 : The dataset is cleaned but proteins are not yet structurally aligned
        # 2 : The dataset is cleaned and proteins are already structurally aligned
	# Note : The status of your dataset guides Voxeler in its data cleaning process

tmalign_reference = None
	# Default : None
	# Possible values :
		# first : will choose the first PDB file (in pymol session) as the reference for the tmalign alignement
		# 'pdb_ref' : will choose the pdb file corresponding to this id as the reference for the tmalign alignement
	# Note : The reference is used to align all the other PDB files, with tmalign plugin
